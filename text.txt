local LogParagraph

local function addLog(amount)
	if not LogParagraph then return end
	local timeStr = os.date("%H:%M:%S")
	table.insert(logs, 1, string.format("[%s] +%s원", timeStr, tostring(amount)))

	if #logs > MAX_LOGS then
		table.remove(logs)
	end

	LogParagraph:Set({
		Title = "잔액 변동 기록",
		Content = table.concat(logs, "\n"),
	})
end

local function updateTimeLabel()
	local elapsed = os.time() - scriptStartTime
	local minutes = math.floor(elapsed / 60)
	local seconds = elapsed % 60
	TimeLabel:Set(string.format("적용 시간: %d분 %d초", minutes, seconds))
end

	while true do
		local currentMoney = 0
		if player:FindFirstChild("leaderstats") and player.leaderstats:FindFirstChild("Money") then
			currentMoney = player.leaderstats.Money.Value
		end
		MoneyLabel:Set("돈: " .. tostring(currentMoney))

			local gained = currentMoney - lastMoney
			totalEarned = totalEarned + gained
			EarnedLabel:Set("총 수익: " .. tostring(totalEarned))

			Rayfield:Notify({
				Title = "잔액 변동 알림",
				Content = "+" .. tostring(gained) .. "원",

			})

			addLog(gained)

		lastMoney = currentMoney

		local jobText = "없음"
		if player:FindFirstChild("leaderstats") and player.leaderstats:FindFirstChild("Job") then
			jobText = tostring(player.leaderstats.Job.Value)
		elseif player.Team then
			jobText = player.Team.Name
		end

		if string.find(jobText, " \\ ") then
			jobText = string.split(jobText, " \\ ")[1]

		JobLabel:Set("직업: " .. jobText)

		task.wait(0.3)
	end

local BIKER_KEYS  = { "폭주족", "biker" }

local function inTeam(plr, keys)
	local t = plr and plr.Team
	if not t then return false end
	local n = string.lower(t.Name)
	for _, k in ipairs(keys) do
		if string.find(n, k, 1, true) then
			return true
		end
	end
	return false
end

local function teamColor(plr)
	local t = plr.Team
	if t and t.TeamColor then
		return t.TeamColor.Color
	end
	return Color3.fromRGB(255, 255, 255)
end

local function getBounty(plr)
	local ls = plr:FindFirstChild("leaderstats")
	if not ls then return nil end
	for _, v in ipairs(ls:GetChildren()) do
		local n = string.lower(v.Name)
		if string.find(n, "현상금", 1, true) or string.find(n, "bounty", 1, true) then
			return v.Value
		end
	end
	return nil
end

local function fmtNum(n)
	if type(n) ~= "number" then return tostring(n) end
	local str = tostring(math.floor(n))
	local rev = string.reverse(str)
	rev = string.gsub(rev, "(%d%d%d)", "%1,")
	str = string.reverse(rev)
	str = string.gsub(str, "^,", "")
	return str
end

local espObjects = {}

local function removeESP(plr)
	local obj = espObjects[plr]
	if obj then
		if obj.Highlight then obj.Highlight:Destroy() end
		if obj.Billboard then obj.Billboard:Destroy() end
		espObjects[plr] = nil
	end
end

local function createESP(plr, char, root)
	removeESP(plr)

	local hl = Instance.new("Highlight")
	hl.Name = "ESP_HL_" .. plr.Name
	hl.Adornee = char
	hl.FillTransparency = 0.5
	hl.OutlineTransparency = 0
	hl.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
	hl.Enabled = false
	hl.Parent = espFolder

	local bb = Instance.new("BillboardGui")
	bb.Name = "ESP_BB_" .. plr.Name
	bb.Adornee = root
	bb.Size = UDim2.new(0, 220, 0, 60)
	bb.StudsOffset = Vector3.new(0, 4, 0)
	bb.AlwaysOnTop = true
	bb.Enabled = false
	bb.Parent = espFolder

	local label = Instance.new("TextLabel")
	label.BackgroundTransparency = 1
	label.Size = UDim2.new(1, 0, 1, 0)
	label.Font = Enum.Font.GothamBold
	label.TextSize = 14
	label.TextStrokeTransparency = 0
	label.Text = ""
	label.Parent = bb

	espObjects[plr] = { Highlight = hl, Billboard = bb, Label = label }
	return espObjects[plr]
end

RunService.RenderStepped:Connect(function()
	if not ESP.Enabled then return end

	local myChar = player.Character
	local myRoot = myChar and myChar:FindFirstChild("HumanoidRootPart")

	for _, plr in ipairs(Players:GetPlayers()) do
		if plr ~= player then
			local char = plr.Character
			local root = char and char:FindFirstChild("HumanoidRootPart")

			if root and inTeam(plr, BIKER_KEYS) then
				local obj = espObjects[plr]
				if not obj or obj.Highlight.Adornee ~= char or obj.Billboard.Adornee ~= root then
					obj = createESP(plr, char, root)
				end

				local dist = myRoot and (myRoot.Position - root.Position).Magnitude or 0
				local visible = dist <= ESP.MaxDistance
				local col = teamColor(plr)

				obj.Highlight.FillColor = col
				obj.Highlight.OutlineColor = col
				obj.Highlight.Enabled = visible and ESP.Highlight
				obj.Label.TextColor3 = col

				local lines = {}
				if ESP.Names then
					table.insert(lines, plr.DisplayName)
				end
				if ESP.Bounty then
					local b = getBounty(plr)
					table.insert(lines, "현상금: " .. (b and fmtNum(b) or "?"))
				end
				if ESP.Distance then
					table.insert(lines, string.format("[%d m]", dist))
				end
				obj.Label.Text = table.concat(lines, "\n")
				obj.Billboard.Enabled = visible and #lines > 0
			else
				removeESP(plr)
			end
		end
	end
end)

local Hitbox = {
	Enabled = false,
	Size = 10,
	Transparency = 0.6,
	Visible = true,
	Color = Color3.fromRGB(255, 255, 0),
}

local originalData = {}
local hitVisuals = {}

local function removeVisual(plr)
	if hitVisuals[plr] then
		hitVisuals[plr]:Destroy()
		hitVisuals[plr] = nil
	end
end

local function restoreHitbox(plr)
	removeVisual(plr)
	local data = originalData[plr]
	if not data then return end
	local char = plr.Character
	local root = char and char:FindFirstChild("HumanoidRootPart")
	if root then
		pcall(function()
			root.Size = data.Size
			root.CanCollide = data.CanCollide
			root.Massless = data.Massless
		end)
	end
	originalData[plr] = nil
end

local function ensureVisual(plr, root)
	local box = hitVisuals[plr]
	if not box or not box.Parent or box.Adornee ~= root then
		removeVisual(plr)
		box = Instance.new("BoxHandleAdornment")
		box.Name = "HB_" .. plr.Name
		box.Adornee = root
		box.AlwaysOnTop = true
		box.ZIndex = 5
		box.Parent = espFolder
		hitVisuals[plr] = box
	end
	box.Size = root.Size
	box.Color3 = Hitbox.Color
	box.Transparency = Hitbox.Transparency
	box.Visible = Hitbox.Visible
end

RunService.Heartbeat:Connect(function()
	local active = Hitbox.Enabled and inTeam(player, POLICE_KEYS)

	for _, plr in ipairs(Players:GetPlayers()) do
		if plr ~= player then
			local char = plr.Character
			local root = char and char:FindFirstChild("HumanoidRootPart")

			if root and active and inTeam(plr, BIKER_KEYS) then
				if not originalData[plr] then
					originalData[plr] = {
						Size = root.Size,
						CanCollide = root.CanCollide,
						Massless = root.Massless,
					}
				end
				pcall(function()
					root.Size = Vector3.new(Hitbox.Size, Hitbox.Size, Hitbox.Size)
					root.CanCollide = false
					root.Massless = true
				end)
				ensureVisual(plr, root)
			elseif originalData[plr] or hitVisuals[plr] then
				restoreHitbox(plr)
			end
		end
	end
end)

Players.PlayerRemoving:Connect(function(plr)
	removeVisual(plr)

local pathDots = {}

local function clearDots()
	for _, d in ipairs(pathDots) do
		d:Destroy()
	end
	pathDots = {}
end

local function drawDots(points)
	clearDots()
	if not GPS.ShowPath then return end
	local step = math.max(1, math.ceil(#points / 60))
	for i = 1, #points, step do
		local dot = Instance.new("SphereHandleAdornment")
		dot.Adornee = Workspace.Terrain
		dot.CFrame = CFrame.new(points[i] + Vector3.new(0, 1.5, 0))
		dot.Radius = 0.6
		dot.Color3 = Color3.fromRGB(0, 200, 255)
		dot.Transparency = 0.2
		dot.AlwaysOnTop = true
		dot.ZIndex = 3
		dot.Parent = espFolder
		table.insert(pathDots, dot)
	end
end

local function getTarget()
	if not GPS.TargetName then return nil end
	local plr = Players:FindFirstChild(GPS.TargetName)
	local char = plr and plr.Character
	local root = char and char:FindFirstChild("HumanoidRootPart")
	return plr, root
end

task.spawn(function()
	while true do
		task.wait(2)
		if GPS.Enabled then
			local plr, targetRoot = getTarget()
			local myChar = player.Character
			local myRoot = myChar and myChar:FindFirstChild("HumanoidRootPart")

			if targetRoot and myRoot then
				local pts = {}
				local ok = pcall(function()
					local path = PathfindingService:CreatePath({
						AgentRadius = 2,
						AgentHeight = 5,
						AgentCanJump = true,
					})
					path:ComputeAsync(myRoot.Position, targetRoot.Position)
					if path.Status == Enum.PathStatus.Success then
						for _, wp in ipairs(path:GetWaypoints()) do
							table.insert(pts, wp.Position)
						end
					end
				end)
				if ok and #pts > 0 then
					GPS.Waypoints = pts
					drawDots(pts)
				else
					GPS.Waypoints = {}
					clearDots()
				end
			else
				GPS.Waypoints = {}
				clearDots()
			end
		end
	end
end)

RunService.RenderStepped:Connect(function()
	if not GPS.Enabled then return end

	local plr, targetRoot = getTarget()
	local myChar = player.Character
	local myRoot = myChar and myChar:FindFirstChild("HumanoidRootPart")
	local cam = Workspace.CurrentCamera

	if not (plr and targetRoot and myRoot and cam) then
		infoLabel.Text = "대상을 찾을 수 없음"
		return
	end

	local aim = targetRoot.Position
	for _, pos in ipairs(GPS.Waypoints) do
		local flatDiff = Vector3.new(pos.X - myRoot.Position.X, 0, pos.Z - myRoot.Position.Z)
		if flatDiff.Magnitude > 8 then
			aim = pos
			break
		end
	end

	local dir = Vector3.new(aim.X - myRoot.Position.X, 0, aim.Z - myRoot.Position.Z)
	local look = Vector3.new(cam.CFrame.LookVector.X, 0, cam.CFrame.LookVector.Z)
	local right = Vector3.new(cam.CFrame.RightVector.X, 0, cam.CFrame.RightVector.Z)

	if dir.Magnitude > 0.1 and look.Magnitude > 0.01 and right.Magnitude > 0.01 then
		dir = dir.Unit
		look = look.Unit
		right = right.Unit
		arrow.Rotation = math.deg(math.atan2(dir:Dot(right), dir:Dot(look)))
	end

	local total = (targetRoot.Position - myRoot.Position).Magnitude
	infoLabel.Text = string.format("%s  [%d m]", plr.DisplayName, total)
end)

	for _, k in ipairs(CUFF_KEYS) do
		if string.find(text, k, 1, true) then
			return true
		end
	end
	return false
end

	end

local skipped = {}
local HANDCUFF_KEYS = { "handcuff", "cuff", "수갑" }
local ArrestToggle

local function findHandcuffs()
	local backpack = player:FindFirstChildOfClass("Backpack")
	for _, container in ipairs({ player.Character, backpack }) do
		if container then
			for _, item in ipairs(container:GetChildren()) do
				if item:IsA("Tool") then
					local n = string.lower(item.Name)
					for _, k in ipairs(HANDCUFF_KEYS) do
						if string.find(n, k, 1, true) then
							return item
						end
					end
				end
			end
		end
	end
	return nil
end

local function equipHandcuffs()
	local tool = findHandcuffs()
	if not tool then return nil end
	local char = player.Character
	local hum = char and char:FindFirstChildOfClass("Humanoid")
	if hum and tool.Parent ~= char then
		pcall(function() hum:EquipTool(tool) end)
	end
	return tool
end

local VIM
pcall(function() VIM = game:GetService("VirtualInputManager") end)

local shownPrompts = {}
ProximityPromptService.PromptShown:Connect(function(prompt)
	shownPrompts[prompt] = true
end)
ProximityPromptService.PromptHidden:Connect(function(prompt)
	shownPrompts[prompt] = nil
end)

local function findPrompt(targetChar)
	for prompt in pairs(shownPrompts) do
		if prompt.Parent and (prompt:IsDescendantOf(targetChar) or isCuffPrompt(prompt)) then
			return prompt
		end
	end
	for _, d in ipairs(targetChar:GetDescendants()) do
		if d:IsA("ProximityPrompt") and d.Enabled then
			return d
		end
	end
	return nil
end

local function triggerPrompt(prompt)
	pcall(function() prompt.HoldDuration = 0 end)

	if fireproximityprompt then
		pcall(fireproximityprompt, prompt)
	end

	if VIM then
		local key = prompt.KeyboardKeyCode
		if key == Enum.KeyCode.Unknown then
			key = Enum.KeyCode.E
		end
		pcall(function()
			VIM:SendKeyEvent(true, key, false, game)
			task.wait(0.15)
			VIM:SendKeyEvent(false, key, false, game)
		end)
	end
end

local function isArrested(plr, char0)
	if not plr.Parent then return true end
	if plr.Character ~= char0 then return true end
	if not inTeam(plr, BIKER_KEYS) then return true end
	for name, val in pairs(char0:GetAttributes()) do
		local n = string.lower(name)
		if val == true and (string.find(n, "cuff", 1, true) or string.find(n, "arrest", 1, true)) then
			return true
		end
	end
	return false
end

local function teleportBehind(mRoot, tRoot)
	local behind = (tRoot.CFrame * CFrame.new(0, 0, AutoArrest.Offset)).Position
	pcall(function()
		mRoot.CFrame = CFrame.lookAt(behind, tRoot.Position)
		mRoot.AssemblyLinearVelocity = Vector3.zero
	end)
end

local function tryEquip()
	for _ = 1, 8 do
		local tool = equipHandcuffs()
		if not tool then return nil end
		task.wait(0.2)
		if tool.Parent == player.Character then
			return tool
		end
	end
	return nil
end

local function arrestLoop()
	AutoArrest.Running = true
	local done, failed = 0, 0
	local tried = {}

	while AutoArrest.Active do
		if not inTeam(player, POLICE_KEYS) then
			Rayfield:Notify({ Title = "Auto Arrest", Content = "경찰팀이 아님", Duration = 3 })
			break
		end
		if not findHandcuffs() then
			Rayfield:Notify({ Title = "Auto Arrest", Content = "수갑을 찾을 수 없음", Duration = 3 })
			break
		end

		local myRoot = player.Character and player.Character:FindFirstChild("HumanoidRootPart")
		if not myRoot then
			task.wait(0.5)
			continue
		end

		local list = {}
		for _, plr in ipairs(Players:GetPlayers()) do
			if plr ~= player and not tried[plr] and inTeam(plr, BIKER_KEYS) then
				local root = plr.Character and plr.Character:FindFirstChild("HumanoidRootPart")
				if root then
					table.insert(list, { plr = plr, dist = (root.Position - myRoot.Position).Magnitude })
				end
			end
		end
		table.sort(list, function(a, b) return a.dist < b.dist end)

		if #list == 0 then break end

		local target = list[1].plr
		tried[target] = true
		local char0 = target.Character
		local tRoot = char0 and char0:FindFirstChild("HumanoidRootPart")
		if not tRoot then
			failed = failed + 1
			continue
		end

		teleportBehind(myRoot, tRoot)
		task.wait(0.3)

		local tool = tryEquip()
		if not tool then
			failed = failed + 1
			Rayfield:Notify({
				Title = "Auto Arrest",
				Content = target.DisplayName .. ": 수갑을 들 수 없어 건너뜀",
				Duration = 3,
			})
			continue
		end

		local t0 = os.clock()
		local arrested = false
		while AutoArrest.Active and os.clock() - t0 < AutoArrest.Timeout do
			if isArrested(target, char0) then
				arrested = true
				break
			end

			tRoot = char0:FindFirstChild("HumanoidRootPart")
			local mRoot = player.Character and player.Character:FindFirstChild("HumanoidRootPart")
			if not (tRoot and mRoot) then break end

			if (mRoot.Position - tRoot.Position).Magnitude > AutoArrest.Offset + 4 then
				teleportBehind(mRoot, tRoot)
			end

			tool = equipHandcuffs() or tool
			local prompt = findPrompt(char0)
			if prompt then
				triggerPrompt(prompt)
			else
				pcall(function() tool:Activate() end)
			end
			task.wait(0.5)
		end

		if arrested then
			done = done + 1
		else
			failed = failed + 1
		end
		task.wait(0.5)
	end

	AutoArrest.Running = false
	AutoArrest.Active = false
	pcall(function() ArrestToggle:Set(false) end)
	Rayfield:Notify({
		Title = "Auto Arrest",
		Content = string.format("종료 - 성공 %d / 건너뜀·실패 %d", done, failed),
		Duration = 4,
	})
end
