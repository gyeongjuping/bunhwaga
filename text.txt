local LogParagraph

local function addLog(amount)
	if not LogParagraph then return end
	local timeStr = os.date("%H:%M:%S")

	if #logs >= MAX_LOGS then
		logs = {}
	end

	table.insert(logs, 1, string.format("[%s] +%s원", timeStr, tostring(amount)))

	LogParagraph:Set({
		Title = "잔액 변동 기록",
		Content = table.concat(logs, "\n"),
	})
end

local function updateTimeLabel()
	local elapsed = os.time() - scriptStartTime
	local hours = math.floor(elapsed / 3600)
	local minutes = math.floor((elapsed % 3600) / 60)
	TimeLabel:Set(string.format("적용 시간: %d시간 %d분", hours, minutes))
end

	while true do
		local currentMoney = 0
		if player:FindFirstChild("leaderstats") and player.leaderstats:FindFirstChild("Money") then
			currentMoney = player.leaderstats.Money.Value
		end
		MoneyLabel:Set("돈: " .. tostring(currentMoney))

			totalEarned = totalEarned + gained

				Content = "+" .. tostring(gained) .. "원",

		local jobText = "없음"
		if player:FindFirstChild("leaderstats") and player.leaderstats:FindFirstChild("Job") then
			jobText = tostring(player.leaderstats.Job.Value)
		elseif player.Team then
			jobText = player.Team.Name
		end

		if string.find(jobText, " \\ ") then
			jobText = string.split(jobText, " \\ ")[1]
		elseif string.find(jobText, "ㅣ") then

			jobText = string.split(jobText, " | ")[1]

		JobLabel:Set("직업: " .. jobText)

local BIKER_KEYS  = { "폭주족", "biker" }
local POLICE_KEYS = { "경찰", "police" }

local function inTeam(plr, keys)
	local t = plr and plr.Team
	if not t then return false end
	local n = string.lower(t.Name)
	for _, k in ipairs(keys) do
		if string.find(n, k, 1, true) then
			return true
		end
	end
	return false
end

local function teamColor(plr)
	local t = plr.Team
	if t and t.TeamColor then
		return t.TeamColor.Color
	end
	return Color3.fromRGB(255, 255, 255)
end

local function getBounty(plr)
	local ls = plr:FindFirstChild("leaderstats")
	if not ls then return nil end
	for _, v in ipairs(ls:GetChildren()) do
		local n = string.lower(v.Name)
		if string.find(n, "현상금", 1, true) or string.find(n, "bounty", 1, true) then
			return v.Value
		end
	end
	return nil
end

local function fmtNum(n)
	if type(n) ~= "number" then return tostring(n) end
	local str = tostring(math.floor(n))
	local rev = string.reverse(str)
	rev = string.gsub(rev, "(%d%d%d)", "%1,")
	str = string.reverse(rev)
	str = string.gsub(str, "^,", "")
	return str
end

	MaxDistance = 3000,

local espObjects = {}

local function removeESP(plr)
	local obj = espObjects[plr]
	if obj then
		if obj.Billboard then obj.Billboard:Destroy() end
		espObjects[plr] = nil
	end
end

local function createESP(plr, root)
	removeESP(plr)

	local bb = Instance.new("BillboardGui")
	bb.Name = "ESP_BB_" .. plr.Name
	bb.Adornee = root
	bb.Size = UDim2.new(0, 220, 0, 60)
	bb.StudsOffset = Vector3.new(0, 4, 0)
	bb.AlwaysOnTop = true
	bb.Enabled = false
	bb.Parent = espFolder

	local label = Instance.new("TextLabel")
	label.BackgroundTransparency = 1
	label.Size = UDim2.new(1, 0, 1, 0)
	label.Font = Enum.Font.GothamBold
	label.TextSize = 14
	label.TextStrokeTransparency = 0
	label.Text = ""
	label.Parent = bb

	espObjects[plr] = { Billboard = bb, Label = label }
	return espObjects[plr]
end

	if not ESP.Enabled then return end

	local myChar = player.Character
	local myRoot = myChar and myChar:FindFirstChild("HumanoidRootPart")

	for _, plr in ipairs(Players:GetPlayers()) do
		if plr ~= player then
			local char = plr.Character
			local root = char and char:FindFirstChild("HumanoidRootPart")

			if root and inTeam(plr, BIKER_KEYS) then
				local obj = espObjects[plr]
				if not obj or obj.Billboard.Adornee ~= root then
					obj = createESP(plr, root)
				end

				local dist = myRoot and (myRoot.Position - root.Position).Magnitude or 0
				obj.Billboard.Enabled = dist <= ESP.MaxDistance
				obj.Label.TextColor3 = teamColor(plr)

				local now = os.clock()
				if now - (obj.lastUpdate or 0) >= 1 then
					obj.lastUpdate = now
					local b = getBounty(plr)
					obj.Label.Text = string.format(
						"%s\n현상금: %s\n[%d m]",
						plr.DisplayName,
						b and fmtNum(b) or "?",
						dist
					)
				end
			else
				removeESP(plr)
			end
		end
	end
end)

local Hitbox = {
	Enabled = false,
	Size = 10,
	Transparency = 0.6,
	Visible = true,
	Color = Color3.fromRGB(255, 255, 0),
}

local originalData = {}
local hitVisuals = {}

local function removeVisual(plr)
	if hitVisuals[plr] then
		hitVisuals[plr]:Destroy()
		hitVisuals[plr] = nil
	end
end

local function restoreHitbox(plr)
	removeVisual(plr)
	local data = originalData[plr]
	if not data then return end
	local char = plr.Character
	local root = char and char:FindFirstChild("HumanoidRootPart")
	if root then
		pcall(function()
			root.Size = data.Size
			root.CanCollide = data.CanCollide
			root.Massless = data.Massless
		end)
	end
	originalData[plr] = nil
end

local function ensureVisual(plr, root)
	local box = hitVisuals[plr]
	if not box or not box.Parent or box.Adornee ~= root then
		removeVisual(plr)
		box = Instance.new("BoxHandleAdornment")
		box.Name = "HB_" .. plr.Name
		box.Adornee = root
		box.AlwaysOnTop = true
		box.ZIndex = 5
		box.Parent = espFolder
		hitVisuals[plr] = box
	end
	box.Size = root.Size
	box.Color3 = Hitbox.Color
	box.Transparency = Hitbox.Transparency
	box.Visible = Hitbox.Visible
end

RunService.Heartbeat:Connect(function()
	local active = Hitbox.Enabled and inTeam(player, POLICE_KEYS)

	for _, plr in ipairs(Players:GetPlayers()) do
		if plr ~= player then
			local char = plr.Character
			local root = char and char:FindFirstChild("HumanoidRootPart")

			if root and active and inTeam(plr, BIKER_KEYS) then
				if not originalData[plr] then
					originalData[plr] = {
						Size = root.Size,
						CanCollide = root.CanCollide,
						Massless = root.Massless,
					}
				end
				pcall(function()
					root.Size = Vector3.new(Hitbox.Size, Hitbox.Size, Hitbox.Size)
					root.CanCollide = false
					root.Massless = true
				end)
				ensureVisual(plr, root)
			elseif originalData[plr] or hitVisuals[plr] then
				restoreHitbox(plr)
			end
		end
	end
end)

Players.PlayerRemoving:Connect(function(plr)
	removeVisual(plr)
	originalData[plr] = nil
end)

local CUFF_KEYS = { "수갑", "handcuff", "체포", "arrest" }
local originalHold = setmetatable({}, { __mode = "k" })

local function isCuffPrompt(prompt)
	local text = string.lower(
		tostring(prompt.ActionText) .. " " ..
		tostring(prompt.ObjectText) .. " " ..
		prompt.Name
	)
	for _, k in ipairs(CUFF_KEYS) do
		if string.find(text, k, 1, true) then
			return true
		end
	end
	return false
end

local function onPlayerCharacter(prompt)
	local inst = prompt.Parent
	while inst and inst ~= Workspace do
		if inst:IsA("Model") and Players:GetPlayerFromCharacter(inst) then
			return true
		end
		inst = inst.Parent
	end
	return false
end

ProximityPromptService.PromptShown:Connect(function(prompt)
	if not Police.InstantPrompt then return end
	if not (isCuffPrompt(prompt) or onPlayerCharacter(prompt)) then return end

	if originalHold[prompt] == nil then
		originalHold[prompt] = prompt.HoldDuration
	end
	prompt.HoldDuration = 0
end)

local function restorePrompts()
	for prompt, hold in pairs(originalHold) do
		pcall(function() prompt.HoldDuration = hold end)
		originalHold[prompt] = nil
	end
end

	shownPrompts[prompt] = true
end)

	shownPrompts[prompt] = nil
end)

local function findPrompt(targetChar)
	for prompt in pairs(shownPrompts) do
		if prompt.Parent and (prompt:IsDescendantOf(targetChar) or isCuffPrompt(prompt)) then
			return prompt
		end
	end
	for _, d in ipairs(targetChar:GetDescendants()) do
		if d:IsA("ProximityPrompt") and d.Enabled then
			return d
		end
	end
	return nil
end

local function triggerPrompt(prompt)
	pcall(function() prompt.HoldDuration = 0 end)

	if fireproximityprompt then
		pcall(fireproximityprompt, prompt)
	end

	if VIM then
		local key = prompt.KeyboardKeyCode
		if key == Enum.KeyCode.Unknown then
			key = Enum.KeyCode.E
		end
		pcall(function()
			VIM:SendKeyEvent(true, key, false, game)
			task.wait(0.15)
			VIM:SendKeyEvent(false, key, false, game)
		end)
	end
end

local function isArrested(plr, char0)
	if not plr.Parent then return true end
	if plr.Character ~= char0 then return true end
	if not inTeam(plr, BIKER_KEYS) then return true end
	for name, val in pairs(char0:GetAttributes()) do
		local n = string.lower(name)
		if val == true and (string.find(n, "cuff", 1, true) or string.find(n, "arrest", 1, true)) then
			return true
		end
	end
	return false
end

local ZONE_NAME = "RestrictedZone"
local zoneParts = {}
local lastZoneScan = 0

local function refreshZones(force)
	if not force and os.clock() - lastZoneScan < 60 then return end
	lastZoneScan = os.clock()
	local parts = {}
	local count = 0
	for _, d in ipairs(Workspace:GetDescendants()) do
		count = count + 1
		if count % 3000 == 0 then
			task.wait()
		end
		if d.Name == ZONE_NAME then
			if d:IsA("BasePart") then
				table.insert(parts, d)
			else
				for _, c in ipairs(d:GetDescendants()) do
					if c:IsA("BasePart") then
						table.insert(parts, c)
					end
				end
			end
		end
	end
	zoneParts = parts
end

local function inRestrictedZone(pos)
	for _, part in ipairs(zoneParts) do
		if part.Parent then
			local rel = part.CFrame:PointToObjectSpace(pos)
			local half = part.Size / 2
			if math.abs(rel.X) <= half.X + 2
				and math.abs(rel.Y) <= half.Y + 8
				and math.abs(rel.Z) <= half.Z + 2 then
				return true
			end
		end
	end
	return false
end

local failedZones = {}
local ZONE_RADIUS = 70

local function inFailedZone(pos)
	for _, z in ipairs(failedZones) do
		if (pos - z).Magnitude <= ZONE_RADIUS then
			return true
		end
	end
	return false
end

local function looksSafe(plr)
	local char = plr.Character
	if not char then return true end
	if char:FindFirstChildOfClass("ForceField") then return true end
	for _, obj in ipairs({ char, plr }) do
		for name, val in pairs(obj:GetAttributes()) do
			if val == true and string.find(string.lower(name), "safe", 1, true) then
				return true
			end
		end
	end
	return false
end

local function teleportBehind(mRoot, tRoot)
	local vel = tRoot.AssemblyLinearVelocity
	local back = AutoArrest.Offset
	if vel.Magnitude > 20 then
		back = math.max(back, 2)
	end
	pcall(function()
		mRoot.CFrame = tRoot.CFrame * CFrame.new(0, 0, back)
		mRoot.AssemblyLinearVelocity = vel
	end)
end

	return nil
end

local function arrestLoop()
	AutoArrest.Running = true
	failedZones = {}
	refreshZones(true)
	local done, failed = 0, 0
	local tried = {}
	local zoneCooldown = {}

	while AutoArrest.Active do
		if not inTeam(player, POLICE_KEYS) then
			Rayfield:Notify({ Title = "Auto Arrest", Content = "경찰팀이 아님", Duration = 3 })
			break
		end
		if not findHandcuffs() then
			Rayfield:Notify({ Title = "Auto Arrest", Content = "수갑을 찾을 수 없음", Duration = 3 })
			break
		end

		local myRoot = player.Character and player.Character:FindFirstChild("HumanoidRootPart")
		if not myRoot then
			task.wait(0.5)
			continue
		end

		refreshZones()

		local list = {}
		local zoneSkipped = 0
		for _, plr in ipairs(Players:GetPlayers()) do
			if plr ~= player and not tried[plr] and inTeam(plr, BIKER_KEYS) then
				local root = plr.Character and plr.Character:FindFirstChild("HumanoidRootPart")
				if root then
					local cooling = zoneCooldown[plr] and zoneCooldown[plr] > os.clock()
					if cooling or inRestrictedZone(root.Position) then
						zoneSkipped = zoneSkipped + 1
					elseif not inFailedZone(root.Position) and not looksSafe(plr) then
						table.insert(list, { plr = plr, dist = (root.Position - myRoot.Position).Magnitude })
					end
				end
			end
		end
		table.sort(list, function(a, b) return a.dist < b.dist end)

		if #list == 0 then
			if zoneSkipped > 0 then
				task.wait(1)
				continue
			end
			break
		end

		local target = list[1].plr
		tried[target] = true
		local char0 = target.Character
		local tRoot = char0 and char0:FindFirstChild("HumanoidRootPart")
		if not tRoot then
			failed = failed + 1
			continue
		end

		teleportBehind(myRoot, tRoot)
		task.wait(0.2)

		local tool = tryEquip()
		if not tool then
			table.insert(failedZones, tRoot.Position)
			failed = failed + 1
			Rayfield:Notify({
				Title = "Auto Arrest",
				Content = target.DisplayName .. ": 수갑을 들 수 없어 이 구역 건너뜀",
				Duration = 3,
			})
			continue
		end

		local t0 = os.clock()
		local lastEquip, lastTrigger = 0, 0
		local equipTries = 0
		local arrested, gaveUp, enteredZone = false, false, false

		while AutoArrest.Active and os.clock() - t0 < AutoArrest.Timeout do
			if isArrested(target, char0) then
				arrested = true
				break
			end

			tRoot = char0:FindFirstChild("HumanoidRootPart")
			local mChar = player.Character
			local mRoot = mChar and mChar:FindFirstChild("HumanoidRootPart")
			if not (tRoot and mRoot) then break end

			if inRestrictedZone(tRoot.Position) then
				enteredZone = true
				break
			end

			if AutoArrest.LockOn or (mRoot.Position - tRoot.Position).Magnitude > AutoArrest.Offset + 1.5 then
				teleportBehind(mRoot, tRoot)
			end

			local now = os.clock()
			if tool.Parent ~= mChar then
				if now - lastEquip > 0.12 then
					lastEquip = now
					equipTries = equipTries + 1
					if equipTries > 8 then
						gaveUp = true
						break
					end
					tool = equipHandcuffs() or tool
				end
			else
				equipTries = 0
				if now - lastTrigger > 0.4 then
					lastTrigger = now
					local prompt = findPrompt(char0)
					if prompt then
						task.spawn(triggerPrompt, prompt)
					else
						pcall(function() tool:Activate() end)
					end
				end
			end

			task.wait()
		end

		if arrested then
			done = done + 1
		elseif enteredZone then
			tried[target] = nil
			zoneCooldown[target] = os.clock() + 10
		elseif gaveUp then
			if tRoot then
				table.insert(failedZones, tRoot.Position)
			end
			failed = failed + 1
			Rayfield:Notify({
				Title = "Auto Arrest",
				Content = target.DisplayName .. ": 수갑이 계속 내려가 건너뜀",
				Duration = 3,
			})
		else
			failed = failed + 1
		end
		task.wait(0.3)
	end

	AutoArrest.Running = false
	AutoArrest.Active = false
	Rayfield:Notify({
		Title = "Auto Arrest",
		Content = string.format("종료 - 성공 %d / 건너뜀·실패 %d", done, failed),
		Duration = 4,
	})
end

local DEFAULT_SCRIPT_URL = "https://raw.githubusercontent.com/gyeongjuping/bunhwaga/refs/heads/main/text.txt"

	Mode = "인원 적은 서버",
	Busy = false,
	AutoReload = true,
	ScriptURL = DEFAULT_SCRIPT_URL,

local queueFn = (syn and syn.queue_on_teleport)
	or queue_on_teleport
	or (fluxus and fluxus.queue_on_teleport)

local function queueReload()
	if not ServerHop.AutoReload then return end

	if not queueFn then
		Rayfield:Notify({
			Title = "Server Hop",
			Content = "이 실행기는 자동 재실행을 지원하지 않음",
			Duration = 4,
		})
		return
	end

	if ServerHop.ScriptURL == "" then
		Rayfield:Notify({
			Title = "Server Hop",
			Content = "스크립트 URL이 없어 자동 재실행 불가",
			Duration = 4,
		})
		return
	end

	pcall(function()
		queueFn(string.format("loadstring(game:HttpGet(%q))()", ServerHop.ScriptURL))
	end)
end

local function httpGet(url)
	local ok, res = pcall(function()
		return game:HttpGet(url)
	end)
	if ok and res then return res end

	local req = (syn and syn.request) or request or http_request
	if req then
		local ok2, r = pcall(function()
			return req({ Url = url, Method = "GET" })
		end)
		if ok2 and r and r.Body then
			return r.Body
		end
	end
	return nil
end

local function fetchServers()
	local list = {}
	local cursor = ""

	for _ = 1, 3 do
		local url = string.format(
			"https://games.roblox.com/v1/games/%d/servers/Public?sortOrder=Asc&limit=100%s",
			game.PlaceId,
			cursor ~= "" and ("&cursor=" .. cursor) or ""
		)
		local body = httpGet(url)
		if not body then break end

		local ok, data = pcall(function()
			return HttpService:JSONDecode(body)
		end)
		if not ok or type(data) ~= "table" or type(data.data) ~= "table" then break end

		for _, srv in ipairs(data.data) do
			table.insert(list, srv)
		end

		if data.nextPageCursor then
			cursor = data.nextPageCursor
		else
			break
		end
	end

	return list
end

local function pickServer(list)
	local candidates = {}
	for _, srv in ipairs(list) do
		if srv.id ~= game.JobId
			and type(srv.playing) == "number"
			and type(srv.maxPlayers) == "number"
			and srv.playing < srv.maxPlayers - 1 then
			table.insert(candidates, srv)
		end
	end

	if #candidates == 0 then return nil end

	if ServerHop.Mode == "인원 적은 서버" then
		table.sort(candidates, function(a, b) return a.playing < b.playing end)
	elseif ServerHop.Mode == "인원 많은 서버" then
		table.sort(candidates, function(a, b) return a.playing > b.playing end)
	elseif ServerHop.Mode == "핑 낮은 서버" then
		table.sort(candidates, function(a, b)
			return (a.ping or 9999) < (b.ping or 9999)
		end)
	else
		return candidates[math.random(1, #candidates)]
	end

	return candidates[1]
end

local function doHop()
	if ServerHop.Busy then return end
	ServerHop.Busy = true

	Rayfield:Notify({
		Title = "Server Hop",
		Content = "서버 목록을 불러오는 중...",
		Duration = 2,
	})

	local list = fetchServers()
	if #list == 0 then
		Rayfield:Notify({
			Title = "Server Hop",
			Content = "서버 목록을 불러오지 못함",
			Duration = 3,
		})
		ServerHop.Busy = false
		return
	end

	local pick = pickServer(list)
	if not pick then
		Rayfield:Notify({
			Title = "Server Hop",
			Content = "조건에 맞는 서버가 없음",
			Duration = 3,
		})
		ServerHop.Busy = false
		return
	end

	Rayfield:Notify({
		Title = "Server Hop",
		Content = string.format("이동 중... (%d/%d명, 핑 %s)", pick.playing, pick.maxPlayers, tostring(pick.ping or "?")),
		Duration = 3,
	})

	queueReload()

	local ok = pcall(function()
		TeleportService:TeleportToPlaceInstance(game.PlaceId, pick.id, player)
	end)

	if not ok then
		Rayfield:Notify({
			Title = "Server Hop",
			Content = "이동 실패",
			Duration = 3,
		})
	end

	task.delay(5, function()
		ServerHop.Busy = false
	end)
end

do
	local playerGui = player:WaitForChild("PlayerGui")
	local updating = false

	local function changeText()
		local team = player.Team
		if not team then return end

		for _, topda in ipairs(playerGui:GetDescendants()) do
			if topda.Name == "topda" then
				local background = topda:FindFirstChild("background", true)

				if background then
					local labels = {}

					for _, obj in ipairs(background:GetDescendants()) do
						if obj:IsA("TextLabel") then
							table.insert(labels, obj)
						end
					end

					if team.Name == "경찰 | police" then
						if labels[1] then
							labels[1].Text = "번화가돌기 GUI V3"
						end
						if labels[2] then
							labels[2].Text = "경찰모드"
						end
					elseif team.Name == "폭주족 | biker" then
						if labels[1] then
							labels[1].Text = "번화가돌기 GUI V3"
						end
						if labels[2] then
							labels[2].Text = "폭주모드"
						end
					end
				end
			end
		end
	end

	local function scheduleChange(delayTime)
		if updating then return end
		updating = true
		task.delay(delayTime, function()
			updating = false
			pcall(changeText)
		end)
	end

	task.spawn(function()
		task.wait(1)
		pcall(changeText)
	end)

	player:GetPropertyChangedSignal("Team"):Connect(function()
		scheduleChange(1)
	end)

	playerGui.DescendantAdded:Connect(function()
		scheduleChange(0.5)
	end)
end
